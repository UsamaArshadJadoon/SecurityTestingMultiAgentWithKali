# 2. Setup Kali VM (one-time, 30 minutes)
bash kali-setup/kali-init.sh

# 3. Verify tools
bash kali-setup/verify-tools.sh

# 4. Create engagement
bash scripts/setup-engagement.sh acme-corp

# 5. Add credentials
